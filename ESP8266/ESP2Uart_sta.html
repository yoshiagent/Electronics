//example 1  (使用手機熱點分享)
#include <ESP8266WiFi.h>
#define UART_BAUD 115200
#define ssid  "yourRouterSSID"
#define password   "yourRouterPassward"
#define led  2
#define packTimeout 5 // ms
#define bufferSize 8192
//#define bufferSize 8
#define port 9876   //若使用RoboRemo 需用此Port
WiFiServer server(port);//設定Server port
WiFiClient client;  //連進softAP之client
byte buf1[bufferSize];
byte i1=0;
byte buf2[bufferSize];
byte i2=0;

void setup() {
  pinMode(led,OUTPUT);
  digitalWrite(led,1);
  Serial.begin(UART_BAUD);
  WiFi.begin(ssid, password); //連進router
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  WiFi.mode(WIFI_AP_STA);
  WiFi.softAP(String("IPGGx-"+ String((WiFi.localIP().toString()))),"");
  //藉以取得ESP所分配到的動態IP 打開WiFi 可搜尋到 IP-xxxxxxx
  server.begin();
  digitalWrite(led,0);
}

void loop(){
  if(!client.connected()) { // if client not connected
    client = server.available(); // wait for it to connect
    return;
   }

 service(client);
 }

void service(WiFiClient client){
  // here we have a connected client

  if(client.available()) {
    while(client.available()) {
      buf1[i1] = (byte)client.read(); // read char from client
      if(i1<bufferSize-1) i1++;
    }
    // now send to UART:
    Serial.write(buf1, i1);
    i1 = 0;
  }

  if(Serial.available()) {

    // read the data until pause:
 
    while(1) {
      if(Serial.available()) {
        buf2[i2] = (char)Serial.read(); // read char from UART
        if(i2<bufferSize-1) i2++;
      } else {
        //delayMicroseconds(packTimeoutMicros);
        delay(packTimeout);
        if(!Serial.available()) {
          break;
        }
      }
    }
 
    // now send to WiFi:
    client.flush();
    client.write((char*)buf2, i2);
    i2 = 0;
  }
 }

搜尋基地台
#include "ESP8266WiFi.h"

void setup() {
  Serial.begin(115200);

  // Set WiFi to station mode and disconnect from an AP if it was previously connected
  WiFi.mode(WIFI_STA);
  WiFi.disconnect();
  delay(100);

  Serial.println("Setup done");
}

void loop() {
  Serial.println("scan start");

  // WiFi.scanNetworks will return the number of networks found
  int n = WiFi.scanNetworks();
  Serial.println("scan done");
  if (n == 0) {
    Serial.println("no networks found");
  } else {
    Serial.print(n);
    Serial.println(" networks found");
    for (int i = 0; i < n; ++i) {
      // Print SSID and RSSI for each network found
      Serial.print(i + 1);
      Serial.print(": ");
      Serial.print(WiFi.SSID(i));
      Serial.print(" (");
      Serial.print(WiFi.RSSI(i));
      Serial.print(")");
      Serial.println((WiFi.encryptionType(i) == ENC_TYPE_NONE) ? " " : "*");
      delay(10);
    }
  }
  Serial.println("");

  // Wait a bit before scanning again
  delay(5000);
}
