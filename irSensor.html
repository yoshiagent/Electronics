#define   DirR    12
#define   DirL    11
#define   PwmL   9
#define   PwmR  10
#define   irin   A4
#define   GND   A2
#define   VCC   A3
#define   LED   8
#define   modePin  7
int val;
int mode;
void setup() {
  pinMode( DirL , OUTPUT);
  pinMode( DirR , OUTPUT);
  pinMode( GND , OUTPUT);
  pinMode( VCC , OUTPUT);
  pinMode( LED , OUTPUT);
  pinMode( irin , INPUT);
  pinMode( modePin , INPUT);
  digitalWrite(GND,LOW);
  digitalWrite(VCC,HIGH);
  analogWrite(PwmL,0);
  analogWrite(PwmR,0);
  digitalWrite(DirL,HIGH);
  digitalWrite(DirR,HIGH);
  mode=digitalRead(modePin);
}

void loop() {
  val=digitalRead(irin);
  if(mode==HIGH){ //D7接Vcc
   if(val==HIGH)
  {
    digitalWrite(LED,HIGH);
    analogWrite(PwmL,150);
    analogWrite(PwmR,0);
    }
  else
  {digitalWrite(LED,LOW);
    analogWrite(PwmL,0);
    analogWrite(PwmR,150);
  }
  }

  if(mode==LOW){  //D7 接地   前後
    analogWrite(PwmL,150);
    analogWrite(PwmR,150);
   if(val==HIGH)
  {
    digitalWrite(LED,HIGH);
    digitalWrite(DirL,HIGH);
  digitalWrite(DirR,HIGH);
   
    }
  else
  {digitalWrite(LED,LOW);
    digitalWrite(DirL,LOW);
  digitalWrite(DirR,LOW);
  }
  }
}

//使用移動平均法(moving average)
#define   irin   A1
#define   DirR    12
#define   DirL    11
#define   PwmL   9
#define   PwmR  10
#define   n     5
int ind = 0;
int sum = 0;
int avg = 0;

int val[n];
void setup() {
  //Serial.begin(9600);
  pinMode( DirL , OUTPUT);
  pinMode( DirR , OUTPUT);
  pinMode( PwmL , OUTPUT);
  pinMode( PwmR , OUTPUT);
  digitalWrite(PwmL,1);
  digitalWrite(PwmR,1);
  digitalWrite(DirL,1);
  digitalWrite(DirR,1);
  delay(1000);
  for(int i=0;i<n;i++)
   {
    val[i]=analogRead(irin);
    sum=sum+val[i];
   }
}

void loop() {
  sum=sum-val[ind];
  val[ind]=analogRead(irin);
  sum=sum+val[ind];
  ind++;
  if (ind>=n)  ind=0;
  avg=sum/n;
  //Serial.println(avg);
  if(avg>500)
  {
      digitalWrite(DirL,1);
   }
  else
  {
    digitalWrite(DirL,0); 
    delay(700);
  }
 
}
