#define Ain  A2
int vin;
int diff;
int vmax;
int vmin;
void setup() {
Serial.begin(9600);
}

void loop() {
    vmax=0;
    vmin=1026;
    for(int i=0;i<200;i++){
     vin=analogRead(Ain);
     if(vin>vmax) vmax=vin;
     if(vin<vmin) vmin=vin;
    }
    diff=vmax-vmin;
     Serial.print("vmax=");
     Serial.print(vmax);
     Serial.print("  vmin=");
     Serial.print(vmin);
     Serial.print("  diff=");
     Serial.println(diff);
     delay(500);
}

// 3. LCD
#define Ain  A2
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
LiquidCrystal_I2C lcd(0x3F,16,2);
int vin;
int diff;
int vmax;
int vmin;
void setup() {
lcd.init(); // initialize the lcd
lcd.backlight();
lcd.setCursor(0, 0) ;

}

void loop() {
    vmax=0;
    vmin=1026;
    for(int i=0;i<200;i++){
     vin=analogRead(Ain);
     if(vin>vmax) vmax=vin;
     if(vin<vmin) vmin=vin;
    }
    diff=vmax-vmin;
     lcd.clear();
     lcd.setCursor(0, 0) ;
     lcd.print(diff);
     if(diff>100){   //可調靈敏度
      lcd.setCursor(0, 1) ;
      lcd.print("       ");
      lcd.setCursor(0, 1) ;
      lcd.print(diff);
      delay(3000);
      }
     delay(200);
}
