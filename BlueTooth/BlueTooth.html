  if(Serial.available()>0)
{
  if(Serial.read()==0x21) //判斷是否為起始字元
  {
   //delay(10);//若資料傳遞不正確 可再加長
   // Wait until data is received  (不用delay)
   while ( !(UCSR0A & (1 << RXC0)) )  ;
   data.b[0]=Serial.read();
  
//delay(10);//若資料傳遞不正確 可再加長
  // Wait until data is received  (不 用delay) 
   while ( !(UCSR0A & (1 << RXC0)) )  ;
   data.b[1]=Serial.read();
   if(data.value==0x3456)
    { //Toggle D12
        digitalWrite(Led,1-digitalRead(Led));
      //asm("SBI 0x03,4");
    }
  }
 
  }
 }

//紅外線感測 與 藍芽遙控  (練習程式合併)
//#define 與 #include 區
#define led  6
#define led1  7
#define irIn A3
//宣告區
int  inB;
int val;
//setup() 設定區:只執行一次
void setup() {
  Serial.begin(115200);
  pinMode(led,OUTPUT) ;
  digitalWrite(led,LOW);
  pinMode(led1,OUTPUT) ;
  digitalWrite(led1,LOW);
}
//loop() 無限迴圈區
void loop() {
    val=analogRead(irIn);
    if(val>500)
      digitalWrite(led1,LOW);
    else
      digitalWrite(led1,HIGH);
   if (Serial.available() > 0) {
     inB = Serial.read();
     if (inB=='1') digitalWrite(led,HIGH);//LED ON
     if (inB=='0') digitalWrite(led,LOW); //LED OFF
   }
}
