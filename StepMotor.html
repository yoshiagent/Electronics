void stepMrev(int steps,int dt)
{
 for(int k=0;k<steps;k++)
  {
    for (int i=3;i>=0;i--)
    {
       for(int j=0;j<4;j++)
        {
          digitalWrite(pin[j],out[i][j]);
         }
       delay(dt);
    }
  }
}

{
  Serial.begin(2400);
  for(int i=0;i<4;i++)
  pinMode( pin[i] , OUTPUT);
}

{
  if (Serial.available() > 0) {
     irin = Serial.read();
     if (irin==0x11)  stepM(512,5); //one turn 2048/4=512
     if (irin==0x21)  stepMrev(512,5); //rev
   }
}
