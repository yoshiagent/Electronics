#define  led  11
int val;
void setup() {
 Serial.begin(9600);
 pinMode(led,OUTPUT);
}

void loop() {
  if(Serial.available()>0)
   { val=Serial.read();
     if(val=='0')
       digitalWrite(led,LOW);
     else if(val=='1')
       digitalWrite(led,HIGH);
   }
}
