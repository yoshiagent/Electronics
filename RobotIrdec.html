#define   DirR    12
#define   DirL    11
#define   PwmL   9
#define   PwmR  10
int irin;

void cM(int dL,int dR,int sL,int sR){
  digitalWrite(DirL,dL);
  digitalWrite(DirR,dR);
  analogWrite(PwmL,sL);
  analogWrite(PwmR,sR);
}

void setup()
{
  Serial.begin(2400);
  pinMode( DirL , OUTPUT);
  pinMode( DirR , OUTPUT);
  cM(HIGH,HIGH,0,0);

}

void loop()
{
  if (Serial.available() > 0) {
     irin= Serial.read();
     if (irin==0x62) cM(HIGH,HIGH,0,0); //停
     if (irin==0x52) cM(HIGH,HIGH,255,255);//前進
     if (irin==0x72) cM(LOW,LOW,255,255);//後退
     if (irin==0x61) cM(LOW,HIGH,255,255);//順時針
     if (irin==0x63) cM(HIGH,LOW,255,255);//逆時針
   }
}
