#define   LED  11
void setup() {
  pinMode(LED,OUTPUT);
for (int   k=0;  k<2;  k=k+1)
    {    
      digitalWrite(LED,HIGH);
      delay(1000);
      digitalWrite(LED,LOW);
      delay(1000);
    }
}
void loop() {
}
