#define  VRxL  A3
#define  DirL   11
#define  PwmL   9
int xL;

void   setup(){
   pinMode(DirL,OUTPUT);
   digitalWrite(DirL,HIGH);
   analogWrite(PwmL,0);
}

void loop(){
   xL=analogRead(VRxL);
   xL=map(xL,0,1023,255,-255);
   if (xL>50)     //前進
   {
     digitalWrite(DirL,HIGH);
     analogWrite(PwmL,xL);
   }
   else if (xL<-50)  //後退
   {
    digitalWrite(DirL,LOW);
    analogWrite(PwmL,-xL);
   }
   else    //停
   analogWrite(PwmL,0);
   delay(100);
}
