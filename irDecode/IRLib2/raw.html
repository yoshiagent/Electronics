NEC protocol






NEC IR transmission frame
For an address or command the LSB is transmitted first and MSB at last. When a key pressed in the remote following bits transmitted sequentially.
(1) Header
     a 9ms (16x562.5us)leading pulse burst + a 4.5ms space
(2) 8 Bit Address
(3) 8 Bit logical inverse of Address*
(4) 8 Bit Command
(5) 8 Bit logical inverse of Command
(6) Termination pulse.  (a 562.5µs pulse burst to mark the end of the space)

*
Extended Mode : NEC address can be extended upto 16 bit if reliability is not a concern. This is done by utilizing both the address and inverse address field. But in this case total message length may not be same. In this way the number of different device address can be upto 65536.


REPEAT CODES:




a 9ms leading pulse burst
a 2.25ms space
a 562.5µs pulse burst to mark the end of the space (and hence end of the transmitted repeat code).

LiquidCrystal_I2C lcd(0x3F,16,2);
#include <IRLibAll.h>
uint32_t val;
//Create a receiver object to listen on pin 2
IRrecvPCI myReceiver(2);

//Create a decoder object
IRdecode myDecoder;  

void setup() {
  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  myReceiver.enableIRIn(); // Start the receiver
  lcd.setCursor(0, 0);
  lcd.print("Ready to receive");
}

void loop() {
  //Continue looping until you get a complete signal received
  if (myReceiver.getResults()) {
    myDecoder.decode();    //Decode it
    val=myDecoder.value;
    if(val==0xFFFFFFFF)
     {lcd.setCursor(0, 1);
       lcd.print("Reapeat");
      }
     else if(val==0)
      {lcd.clear();
       lcd.setCursor(0, 0);
       lcd.print("unknown protocol");
      }
      else
      {lcd.clear();
       lcd.setCursor(0, 0);
       lcd.print("Code:");
       lcd.setCursor(5, 0);
       lcd.print(val,HEX);
      }
    
      }
    myReceiver.enableIRIn();      //Restart receiver
  }
